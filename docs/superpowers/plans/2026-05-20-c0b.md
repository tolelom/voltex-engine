//! RenderAssets — voltex_renderer::Mesh + MaterialUniform 저장소.
//!
//! C0a 의 phantom-typed `MeshHandle(Handle<MeshAsset>)` 의 id/generation 을
//! 내부 `Handle<voltex_renderer::Mesh>` 와 lockstep 으로 유지한다. add_mesh /
//! add_material 가 유일한 등록 진입점인 한 이 invariant 가 깨지지 않는다.

use voltex_asset::{Assets, Handle};
use voltex_ecs::{MaterialHandle, MeshHandle};
use voltex_renderer::{MaterialUniform, Mesh};

pub struct RenderAssets {
    assets: Assets,
}

impl RenderAssets {
    pub fn new() -> Self {
        Self { assets: Assets::new() }
    }

    pub fn add_mesh(&mut self, mesh: Mesh) -> MeshHandle {
        let h: Handle<Mesh> = self.assets.insert(mesh);
        MeshHandle(Handle::from_raw_parts(h.id(), h.generation()))
    }

    pub fn mesh(&self, h: MeshHandle) -> Option<&Mesh> {
        let raw: Handle<Mesh> = Handle::from_raw_parts(h.0.id(), h.0.generation());
        self.assets.get(raw)
    }

    pub fn add_material(&mut self, m: MaterialUniform) -> MaterialHandle {
        let h: Handle<MaterialUniform> = self.assets.insert(m);
        MaterialHandle(Handle::from_raw_parts(h.id(), h.generation()))
    }

    pub fn material(&self, h: MaterialHandle) -> Option<&MaterialUniform> {
        let raw: Handle<MaterialUniform> = Handle::from_raw_parts(h.0.id(), h.0.generation());
        self.assets.get(raw)
    }
}

impl Default for RenderAssets {
    fn default() -> Self {
        Self::new()
    }
}

//! RenderItem + collect_render_items — World 를 iterate 하여 frame 데이터 생성.

use voltex_ecs::{
    propagate_transforms, Entity, MaterialHandle, MeshHandle, World, WorldTransform,
};
use voltex_math::Mat4;
use voltex_renderer::MaterialUniform;

use crate::assets::RenderAssets;

/// 한 엔티티의 한 프레임 렌더 데이터.
pub struct RenderItem {
    pub entity: Entity,
    pub model: Mat4,
    pub mesh: MeshHandle,
    pub material: MaterialUniform,
}

/// `propagate_transforms` 를 먼저 호출한 뒤 WorldTransform + MeshHandle 이 있는
/// 엔티티를 모은다. Stale MeshHandle, MeshHandle 없는 엔티티는 skip.
/// MaterialHandle 누락/스테일이면 `MaterialUniform::new()` (white default).
pub fn collect_render_items(world: &mut World, assets: &RenderAssets) -> Vec<RenderItem> {
    propagate_transforms(world);

    let mut items = Vec::new();
    let pairs: Vec<(Entity, WorldTransform, MeshHandle)> = world
        .query2::<WorldTransform, MeshHandle>()
        .map(|(e, wt, mh)| (e, *wt, *mh))
        .collect();

    for (entity, wt, mh) in pairs {
        if assets.mesh(mh).is_none() {
            continue;
        }
        let material = world
            .get::<MaterialHandle>(entity)
            .and_then(|h| assets.material(*h).copied())
            .unwrap_or_else(MaterialUniform::new);

        items.push(RenderItem {
            entity,
            model: wt.0,
            mesh: mh,
            material,
        });
    }

    items
}

//! BasicRenderer — mesh_shader.wgsl 기반 ECS-driven 렌더러.
//!
//! pipeline + per-entity dynamic UBO + light buffer + 1×1 default texture
//! 를 모두 owning. `render(...)` 한 메서드로 World 를 그린다.

use bytemuck;
use voltex_ecs::World;
use voltex_math::{Mat4, Vec3};
use voltex_renderer::{
    pipeline::create_mesh_pipeline, CameraUniform, GpuTexture, LightUniform,
};

use crate::assets::RenderAssets;
use crate::items::collect_render_items;

pub struct BasicRenderer {
    pipeline: wgpu::RenderPipeline,
    camera_buffer: wgpu::Buffer,
    light_buffer: wgpu::Buffer,
    camera_light_bind_group: wgpu::BindGroup,
    default_texture: GpuTexture,
    aligned_size: u32,
    max_entities: usize,
}

impl BasicRenderer {
    /// `target_format` 은 render target 의 컬러 포맷 (offscreen viewport 또는 swapchain).
    /// `max_entities` 는 한 프레임에 처리 가능한 최대 entity 수 (UBO 크기 사전 할당).
    pub fn new(
        device: &wgpu::Device,
        queue: &wgpu::Queue,
        target_format: wgpu::TextureFormat,
        max_entities: usize,
    ) -> Self {
        let alignment = device.limits().min_uniform_buffer_offset_alignment;
        let cam_size = std::mem::size_of::<CameraUniform>() as u32;
        let aligned_size = ((cam_size + alignment - 1) / alignment) * alignment;

        let cl_layout = camera_light_bind_group_layout(device);
        let tex_layout = GpuTexture::bind_group_layout(device);

        let camera_buffer = device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("BasicRenderer Camera Dynamic UBO"),
            size: (aligned_size as usize * max_entities) as u64,
            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
            mapped_at_creation: false,
        });

        let light_buffer = device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("BasicRenderer Light UBO"),
            size: std::mem::size_of::<LightUniform>() as u64,
            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
            mapped_at_creation: false,
        });

        let camera_light_bind_group = device.create_bind_group(&wgpu::BindGroupDescriptor {
            label: Some("BasicRenderer Camera+Light BG"),
            layout: &cl_layout,
            entries: &[
                wgpu::BindGroupEntry {
                    binding: 0,
                    resource: wgpu::BindingResource::Buffer(wgpu::BufferBinding {
                        buffer: &camera_buffer,
                        offset: 0,
                        size: wgpu::BufferSize::new(cam_size as u64),
                    }),
                },
                wgpu::BindGroupEntry {
                    binding: 1,
                    resource: light_buffer.as_entire_binding(),
                },
            ],
        });

        let default_texture = GpuTexture::white_1x1(device, queue, &tex_layout);
        let pipeline = create_mesh_pipeline(device, target_format, &cl_layout, &tex_layout);

        Self {
            pipeline,
            camera_buffer,
            light_buffer,
            camera_light_bind_group,
            default_texture,
            aligned_size,
            max_entities,
        }
    }

    /// World 의 모든 MeshHandle 엔티티를 dynamic offset draw 로 렌더링한다.
    /// 호출자는 미리 `wgpu::RenderPassDescriptor` 없이 encoder + target_view +
    /// depth_view 만 넘긴다 (내부에서 begin_render_pass 한다).
    #[allow(clippy::too_many_arguments)]
    pub fn render(
        &self,
        queue: &wgpu::Queue,
        encoder: &mut wgpu::CommandEncoder,
        target_view: &wgpu::TextureView,
        depth_view: &wgpu::TextureView,
        clear_color: wgpu::Color,
        world: &mut World,
        assets: &RenderAssets,
        view_proj: Mat4,
        cam_pos: Vec3,
        light: LightUniform,
    ) {
        let items = collect_render_items(world, assets);
        let n = items.len().min(self.max_entities);
        let aligned = self.aligned_size as usize;

        // staging
        let mut staging = vec![0u8; n * aligned];
        for (i, item) in items.iter().take(n).enumerate() {
            let uniform = CameraUniform {
                view_proj: view_proj.cols,
                model: item.model.cols,
                camera_pos: [cam_pos.x, cam_pos.y, cam_pos.z],
                _padding: 0.0,
            };
            let bytes = bytemuck::bytes_of(&uniform);
            let offset = i * aligned;
            staging[offset..offset + bytes.len()].copy_from_slice(bytes);
        }
        if n > 0 {
            queue.write_buffer(&self.camera_buffer, 0, &staging);
        }
        queue.write_buffer(&self.light_buffer, 0, bytemuck::cast_slice(&[light]));

        let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
            label: Some("BasicRenderer Pass"),
            color_attachments: &[Some(wgpu::RenderPassColorAttachment {
                view: target_view,
                resolve_target: None,
                depth_slice: None,
                ops: wgpu::Operations {
                    load: wgpu::LoadOp::Clear(clear_color),
                    store: wgpu::StoreOp::Store,
                },
            })],
            depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
                view: depth_view,
                depth_ops: Some(wgpu::Operations {
                    load: wgpu::LoadOp::Clear(1.0),
                    store: wgpu::StoreOp::Store,
                }),
                stencil_ops: None,
            }),
            occlusion_query_set: None,
            timestamp_writes: None,
            multiview_mask: None,
        });

        pass.set_pipeline(&self.pipeline);
        pass.set_bind_group(1, &self.default_texture.bind_group, &[]);

        for (i, item) in items.iter().take(n).enumerate() {
            let mesh = assets.mesh(item.mesh).expect("collect_render_items skipped stale handles");
            let dynamic_offset = (i as u32) * self.aligned_size;
            pass.set_bind_group(0, &self.camera_light_bind_group, &[dynamic_offset]);
            pass.set_vertex_buffer(0, mesh.vertex_buffer.slice(..));
            pass.set_index_buffer(mesh.index_buffer.slice(..), wgpu::IndexFormat::Uint32);
            pass.draw_indexed(0..mesh.num_indices, 0, 0..1);
        }
    }
}

fn camera_light_bind_group_layout(device: &wgpu::Device) -> wgpu::BindGroupLayout {
    device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
        label: Some("BasicRenderer Camera+Light BGL"),
        entries: &[
            wgpu::BindGroupLayoutEntry {
                binding: 0,
                visibility: wgpu::ShaderStages::VERTEX | wgpu::ShaderStages::FRAGMENT,
                ty: wgpu::BindingType::Buffer {
                    ty: wgpu::BufferBindingType::Uniform,
                    has_dynamic_offset: true,
                    min_binding_size: wgpu::BufferSize::new(
                        std::mem::size_of::<CameraUniform>() as u64,
                    ),
                },
                count: None,
            },
            wgpu::BindGroupLayoutEntry {
                binding: 1,
                visibility: wgpu::ShaderStages::FRAGMENT,
                ty: wgpu::BindingType::Buffer {
                    ty: wgpu::BufferBindingType::Uniform,
                    has_dynamic_offset: false,
                    min_binding_size: None,
                },
                count: None,
            },
        ],
    })
}

    let mat = MaterialUniform::with_params(entity.base_color, entity.metallic, entity.roughness);
    let mat_h = render_assets.add_material(mat);
    arena_mesh_handles.push(mh);
    arena_mat_handles.push(mat_h);

// Spawn arena entities in ECS
let mut arena_ecs_entities = Vec::new();
for (i, model_m) in models.iter().enumerate() {
    let e = world.spawn();
    // arena_model_matrices 는 이미 적용된 Mat4. position/rotation/scale 분해 어렵다면
    // Transform 대신 직접 WorldTransform 컴포넌트 부여 (propagate 가 root 의 Transform 을 보므로,
    // root 가 Transform 을 가지지 않으면 propagate 가 skip → 다른 방식 필요).
    // 가장 단순한 경로: arena_model_matrices 가 translation-only 라면 위치만 추출.
    // 아니면 임시로 root 에 Identity Transform 부여 + WorldTransform 강제 설정.
    //
    // 검토 후 결정. arena 가 정적이므로 매 프레임 propagate 비용 부담은 없음.
    // 안전한 길: Transform { position: 추출, rotation: 0, scale: 추출 }.
    let pos = extract_translation(model_m);
    let scale = extract_scale(model_m);

    world.add(e, voltex_ecs::MeshHandle(arena_mesh_handles[i].0));
    world.add(e, voltex_ecs::MaterialHandle(arena_mat_handles[i].0));

`extract_translation` / `extract_scale` 헬퍼는 `Mat4` 의 마지막 컬럼 / 첫 3 컬럼 norm 으로:

```rust
fn extract_translation(m: &Mat4) -> Vec3 {
    Vec3::new(m.cols[3][0], m.cols[3][1], m.cols[3][2])
}
fn extract_scale(m: &Mat4) -> Vec3 {
    let sx = (m.cols[0][0].powi(2) + m.cols[0][1].powi(2) + m.cols[0][2].powi(2)).sqrt();
    let sy = (m.cols[1][0].powi(2) + m.cols[1][1].powi(2) + m.cols[1][2].powi(2)).sqrt();
    let sz = (m.cols[2][0].powi(2) + m.cols[2][1].powi(2) + m.cols[2][2].powi(2)).sqrt();
    Vec3::new(sx, sy, sz)
}
```

```rust
fn sync_game_to_ecs(
    game: &mut Game,
    world: &mut voltex_ecs::World,
    sphere_mesh_h: voltex_ecs::MeshHandle,
    projectile_mat_h: voltex_ecs::MaterialHandle,
    enemy_mat_h: voltex_ecs::MaterialHandle,
    player_e: voltex_ecs::Entity,
    player_mat_h: voltex_ecs::MaterialHandle,
) {
    // Player
    world.add(player_e, voltex_ecs::Transform::from_position(
        game.player_pos() + Vec3::new(0.0, 0.5, 0.0),
    ));
    let _ = player_mat_h; // 색 변경은 v1 외

    // Projectiles
    let positions = game.projectile_positions();
    while game.projectile_entities.len() < positions.len() {
        let e = world.spawn();
        world.add(e, voltex_ecs::MeshHandle(sphere_mesh_h.0));
        world.add(e, voltex_ecs::MaterialHandle(projectile_mat_h.0));
        game.projectile_entities.push(e);
    }
    while game.projectile_entities.len() > positions.len() {
        if let Some(e) = game.projectile_entities.pop() {
            world.despawn(e);
        }
    }
    for (i, p) in positions.iter().enumerate() {
        world.add(game.projectile_entities[i],
            voltex_ecs::Transform::from_position_scale(*p, Vec3::new(0.3, 0.3, 0.3)));
    }

    // Enemies — 동일 패턴
    let enemy_positions = game.enemy_positions();
    while game.enemy_entities.len() < enemy_positions.len() {
        let e = world.spawn();
        world.add(e, voltex_ecs::MeshHandle(sphere_mesh_h.0));
        world.add(e, voltex_ecs::MaterialHandle(enemy_mat_h.0));
        game.enemy_entities.push(e);
    }
    while game.enemy_entities.len() > enemy_positions.len() {
        if let Some(e) = game.enemy_entities.pop() {
            world.despawn(e);
        }
    }
    for (i, p) in enemy_positions.iter().enumerate() {
        world.add(game.enemy_entities[i],
            voltex_ecs::Transform::from_position_scale(*p, Vec3::new(0.4, 0.4, 0.4)));
    }
}
```
