```rust
/// Ray vs axis-aligned bounding box (slab method).
/// Returns the smallest non-negative `t` (clamped to 0 if ray starts inside),
/// or `None` if the ray misses or only intersects behind the origin.
///
/// Voltex_physics has an equivalent — this is duplicated here to avoid a
/// voltex_editor → voltex_physics dependency.
pub fn ray_vs_aabb(ray: &voltex_math::Ray, aabb: &voltex_math::AABB) -> Option<f32> {
    let mut t_min = f32::NEG_INFINITY;
    let mut t_max = f32::INFINITY;
    let o = [ray.origin.x, ray.origin.y, ray.origin.z];
    let d = [ray.direction.x, ray.direction.y, ray.direction.z];
    let bmin = [aabb.min.x, aabb.min.y, aabb.min.z];
    let bmax = [aabb.max.x, aabb.max.y, aabb.max.z];
    for i in 0..3 {
        if d[i].abs() < 1e-8 {
            if o[i] < bmin[i] || o[i] > bmax[i] { return None; }
        } else {
            let inv = 1.0 / d[i];
            let mut t1 = (bmin[i] - o[i]) * inv;
            let mut t2 = (bmax[i] - o[i]) * inv;
            if t1 > t2 { std::mem::swap(&mut t1, &mut t2); }
            t_min = t_min.max(t1);
            t_max = t_max.min(t2);
            if t_min > t_max { return None; }
        }
    }
    if t_max < 0.0 { return None; }
    Some(t_min.max(0.0))
}

/// Transform a mesh-local AABB into world space by transforming the 8 corners
/// and rebuilding an axis-aligned AABB. Loose for rotated meshes (OBB picking
/// is out of scope per E1 spec).
pub fn world_aabb_from_local(local: &voltex_math::AABB, world: &Mat4) -> voltex_math::AABB {
    let corners = [
        Vec3::new(local.min.x, local.min.y, local.min.z),
        Vec3::new(local.max.x, local.min.y, local.min.z),
        Vec3::new(local.min.x, local.max.y, local.min.z),
        Vec3::new(local.max.x, local.max.y, local.min.z),
        Vec3::new(local.min.x, local.min.y, local.max.z),
        Vec3::new(local.max.x, local.min.y, local.max.z),
        Vec3::new(local.min.x, local.max.y, local.max.z),
        Vec3::new(local.max.x, local.max.y, local.max.z),
    ];
    let mut min = Vec3::new(f32::INFINITY, f32::INFINITY, f32::INFINITY);
    let mut max = Vec3::new(f32::NEG_INFINITY, f32::NEG_INFINITY, f32::NEG_INFINITY);
    for c in &corners {
        let w = *world * Vec4::new(c.x, c.y, c.z, 1.0);
        let inv_w = if w.w.abs() < 1e-8 { 1.0 } else { 1.0 / w.w };
        let p = Vec3::new(w.x * inv_w, w.y * inv_w, w.z * inv_w);
        min.x = min.x.min(p.x); min.y = min.y.min(p.y); min.z = min.z.min(p.z);
        max.x = max.x.max(p.x); max.y = max.y.max(p.y); max.z = max.z.max(p.z);
    }
    voltex_math::AABB::new(min, max)
}
```

```rust
//! Gizmo — translate / rotate / scale handles for the selected entity.
//!
//! Drag math lives in `translate`, `rotate`, `scale` submodules. Rendering
//! is in `renderer`. State and transitions live here.

pub mod translate;
pub mod rotate;
pub mod scale;
pub mod renderer;

use voltex_ecs::{Entity, Transform, World, WorldTransform};
use voltex_math::{Mat4, Vec3};

pub use renderer::GizmoRenderer;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum GizmoMode {
    Translate,
    Rotate,
    Scale,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum GizmoAxis {
    X,
    Y,
    Z,
}

impl GizmoAxis {
    pub fn dir(self) -> Vec3 {
        match self {
            GizmoAxis::X => Vec3::new(1.0, 0.0, 0.0),
            GizmoAxis::Y => Vec3::new(0.0, 1.0, 0.0),
            GizmoAxis::Z => Vec3::new(0.0, 0.0, 1.0),
        }
    }
    pub fn index(self) -> usize {
        match self { GizmoAxis::X => 0, GizmoAxis::Y => 1, GizmoAxis::Z => 2 }
    }
}

/// Result returned by `end_drag`: caller dispatches `SetTransform` so the
/// drag becomes a single undo entry. `initial` is the Transform value at
/// drag start (caller rolls World back to this so `dispatch` captures the
/// correct inverse), `final_transform` is the value at drag end.
pub struct DragResult {
    pub initial: Transform,
    pub final_transform: Transform,
}

#[derive(Debug, Clone)]
struct DragState {
    axis: GizmoAxis,
    initial_transform: Transform,
    gizmo_origin: Vec3,
    plane_normal: Vec3,
    /// For translate/scale: axis projection of the ray-plane hit at drag start.
    /// For rotate: angle at drag start.
    anchor: f32,
}

pub struct GizmoState {
    pub mode: GizmoMode,
    drag: Option<DragState>,
}

impl GizmoState {
    pub fn new() -> Self {
        Self { mode: GizmoMode::Translate, drag: None }
    }

    pub fn set_mode(&mut self, mode: GizmoMode) {
        if self.drag.is_none() {
            self.mode = mode;
        }
    }

    pub fn is_dragging(&self) -> bool { self.drag.is_some() }

    pub fn active_axis(&self) -> Option<GizmoAxis> {
        self.drag.as_ref().map(|d| d.axis)
    }

    /// World position of the gizmo for the given entity (uses WorldTransform
    /// translation, else Transform.position, else None).
    pub fn gizmo_origin(world: &World, entity: Entity) -> Option<Vec3> {
        if let Some(wt) = world.get::<WorldTransform>(entity) {
            return Some(Vec3::new(wt.0.cols[3][0], wt.0.cols[3][1], wt.0.cols[3][2]));
        }
        world.get::<Transform>(entity).map(|t| t.position)
    }

    /// Try to start a drag. Returns true if a gizmo axis was hit.
    pub fn try_begin_drag(
        &mut self,
        world: &mut World,
        entity: Entity,
        view_proj: &Mat4,
        cam_pos: Vec3,
        rect: (f32, f32, f32, f32), // x, y, w, h
        mouse: (f32, f32),
    ) -> bool {
        if self.drag.is_some() { return false; }
        voltex_ecs::propagate_transforms(world);
        let origin = match Self::gizmo_origin(world, entity) {
            Some(o) => o,
            None => return false,
        };
        let initial = match world.get::<Transform>(entity) {
            Some(t) => *t,
            None => return false,
        };
        let inv = match view_proj.inverse() {
            Some(m) => m,
            None => return false,
        };
        let ndc = crate::picking::ndc_from_viewport(mouse.0, mouse.1, rect.0, rect.1, rect.2, rect.3);
        let ray = crate::picking::ray_from_ndc(&inv, cam_pos, ndc);

        let drag = match self.mode {
            GizmoMode::Translate => translate::try_begin(origin, &ray, cam_pos, initial),
            GizmoMode::Rotate    => rotate::try_begin(origin, &ray, cam_pos, initial),
            GizmoMode::Scale     => scale::try_begin(origin, &ray, cam_pos, initial),
        };
        if let Some(d) = drag {
            self.drag = Some(d);
            true
        } else {
            false
        }
    }

    /// Update the entity's Transform directly from the current mouse position.
    pub fn update_drag(
        &mut self,
        world: &mut World,
        entity: Entity,
        view_proj: &Mat4,
        cam_pos: Vec3,
        rect: (f32, f32, f32, f32),
        mouse: (f32, f32),
    ) {
        let drag = match &self.drag { Some(d) => d.clone(), None => return };
        let inv = match view_proj.inverse() { Some(m) => m, None => return };
        let ndc = crate::picking::ndc_from_viewport(mouse.0, mouse.1, rect.0, rect.1, rect.2, rect.3);
        let ray = crate::picking::ray_from_ndc(&inv, cam_pos, ndc);
        let new_t = match self.mode {
            GizmoMode::Translate => translate::compute_transform(&drag, &ray),
            GizmoMode::Rotate    => rotate::compute_transform(&drag, &ray),
            GizmoMode::Scale     => scale::compute_transform(&drag, &ray),
        };
        if let Some(t) = world.get_mut::<Transform>(entity) {
            *t = new_t;
        }
        let _ = cam_pos;
    }

    /// End the drag. Restores the entity Transform to `initial` and returns
    /// `(initial, final)` so the caller dispatches SetTransform once.
    pub fn end_drag(&mut self, world: &mut World, entity: Entity) -> Option<DragResult> {
        let drag = self.drag.take()?;
        let final_transform = *world.get::<Transform>(entity)?;
        // Roll back so dispatch captures the correct inverse.
        if let Some(t) = world.get_mut::<Transform>(entity) {
            *t = drag.initial_transform;
        }
        Some(DragResult { initial: drag.initial_transform, final_transform })
    }
}

impl Default for GizmoState {
    fn default() -> Self { Self::new() }
}

#[cfg(test)]
mod tests {

    #[test]

        assert_eq!(g.mode, GizmoMode::Rotate);
        g.set_mode(GizmoMode::Scale);
        assert_eq!(g.mode, GizmoMode::Scale);
    }
}
```

```rust
//! Translate gizmo — axis-aligned thin AABB hit detection +
//! drag-plane projection for axis-locked motion.

use voltex_ecs::Transform;
use voltex_math::{AABB, Ray, Vec3};

use super::{DragState, GizmoAxis};

const AXIS_LENGTH: f32 = 1.5;
const AXIS_THICKNESS: f32 = 0.08;
const AXIS_BASE_OFFSET: f32 = 0.15; // gap from origin so the 3 arrows don't overlap

/// Local-space AABB for axis A (centred at origin). Convert to world by
/// translating by `gizmo_origin`.
fn axis_local_aabb(axis: GizmoAxis) -> AABB {
    match axis {
        GizmoAxis::X => AABB::new(
            Vec3::new(AXIS_BASE_OFFSET, -AXIS_THICKNESS, -AXIS_THICKNESS),
            Vec3::new(AXIS_LENGTH, AXIS_THICKNESS, AXIS_THICKNESS),
        ),
        GizmoAxis::Y => AABB::new(
            Vec3::new(-AXIS_THICKNESS, AXIS_BASE_OFFSET, -AXIS_THICKNESS),
            Vec3::new(AXIS_THICKNESS, AXIS_LENGTH, AXIS_THICKNESS),
        ),
        GizmoAxis::Z => AABB::new(
            Vec3::new(-AXIS_THICKNESS, -AXIS_THICKNESS, AXIS_BASE_OFFSET),
            Vec3::new(AXIS_THICKNESS, AXIS_THICKNESS, AXIS_LENGTH),
        ),
    }
}

/// Pick the most camera-facing plane that contains `axis`. Returns the plane
/// normal (orthogonal to axis).
pub fn pick_drag_plane_normal(axis: GizmoAxis, gizmo_origin: Vec3, cam_pos: Vec3) -> Vec3 {
    let a = axis.dir();
    let cam_dir = (cam_pos - gizmo_origin).normalize();
    // Two candidate normals: the other two basis axes.
    let (n1, n2) = match axis {
        GizmoAxis::X => (Vec3::new(0.0, 1.0, 0.0), Vec3::new(0.0, 0.0, 1.0)),
        GizmoAxis::Y => (Vec3::new(1.0, 0.0, 0.0), Vec3::new(0.0, 0.0, 1.0)),
        GizmoAxis::Z => (Vec3::new(1.0, 0.0, 0.0), Vec3::new(0.0, 1.0, 0.0)),
    };
    let d1 = cam_dir.dot(n1).abs();
    let d2 = cam_dir.dot(n2).abs();
    let _ = a;
    if d1 >= d2 { n1 } else { n2 }
}

/// Intersect `ray` with the plane through `origin` with `normal`. Returns the
/// hit point, or None if parallel / behind ray.
fn ray_plane_hit(ray: &Ray, origin: Vec3, normal: Vec3) -> Option<Vec3> {
    let denom = normal.dot(ray.direction);
    if denom.abs() < 1e-6 { return None; }
    let t = (origin - ray.origin).dot(normal) / denom;
    if t < 0.0 { return None; }
    Some(ray.origin + ray.direction * t)
}

/// Test the ray against each axis's thin AABB (translated by gizmo_origin).
/// Returns (axis, hit_t) of the nearest hit, or None.
pub fn hit_axis(gizmo_origin: Vec3, ray: &Ray) -> Option<GizmoAxis> {
    let mut best: Option<(GizmoAxis, f32)> = None;
    for axis in [GizmoAxis::X, GizmoAxis::Y, GizmoAxis::Z] {
        let local = axis_local_aabb(axis);
        let world = AABB::new(local.min + gizmo_origin, local.max + gizmo_origin);
        if let Some(t) = crate::picking::ray_vs_aabb(ray, &world) {
            best = match best {
                Some((_, bt)) if bt <= t => best,
                _ => Some((axis, t)),
            };
        }
    }
    best.map(|(a, _)| a)
}

pub fn try_begin(origin: Vec3, ray: &Ray, cam_pos: Vec3, initial: Transform) -> Option<DragState> {
    let axis = hit_axis(origin, ray)?;
    let plane_normal = pick_drag_plane_normal(axis, origin, cam_pos);
    let hit = ray_plane_hit(ray, origin, plane_normal)?;
    let anchor = (hit - origin).dot(axis.dir());
    Some(DragState {
        axis,
        initial_transform: initial,
        gizmo_origin: origin,
        plane_normal,
        anchor,
    })
}

pub fn compute_transform(drag: &DragState, ray: &Ray) -> Transform {
    let hit = match ray_plane_hit(ray, drag.gizmo_origin, drag.plane_normal) {
        Some(h) => h,
        None => return drag.initial_transform,
    };
    let current = (hit - drag.gizmo_origin).dot(drag.axis.dir());
    let delta = current - drag.anchor;
    let mut t = drag.initial_transform;
    t.position = t.position + drag.axis.dir() * delta;
    t
}
```

```rust
// ── Mesh builders ───────────────────────────────────────────────────────────

const ACTIVE_COLOR: [f32; 3] = [1.0, 1.0, 0.0];
const AXIS_LENGTH: f32 = 1.5;
const AXIS_BASE: f32 = 0.15;
const HEAD_SIZE: f32 = 0.15;
const RING_RADIUS: f32 = 1.2;
const RING_SEGMENTS: usize = 32;
const SCALE_HANDLE: f32 = 0.12;

fn axis_color(axis: GizmoAxis, active: Option<GizmoAxis>) -> [f32; 3] {
    if active == Some(axis) { return ACTIVE_COLOR; }
    match axis {
        GizmoAxis::X => [1.0, 0.2, 0.2],
        GizmoAxis::Y => [0.2, 1.0, 0.2],
        GizmoAxis::Z => [0.2, 0.4, 1.0],
    }
}

fn push_line(verts: &mut Vec<LineVertex>, a: Vec3, b: Vec3, color: [f32; 3]) {
    verts.push(LineVertex { position: [a.x, a.y, a.z], _pad: 0.0, color, _pad2: 0.0 });
    verts.push(LineVertex { position: [b.x, b.y, b.z], _pad: 0.0, color, _pad2: 0.0 });
}

fn build_translate(verts: &mut Vec<LineVertex>, o: Vec3, active: Option<GizmoAxis>) {
    for axis in [GizmoAxis::X, GizmoAxis::Y, GizmoAxis::Z] {
        let color = axis_color(axis, active);
        let dir = axis.dir();
        let start = o + dir * AXIS_BASE;
        let end = o + dir * AXIS_LENGTH;
        push_line(verts, start, end, color);
        // Arrowhead: 4 short crossing lines at the tip.
        let (u, v) = ortho_basis(dir);
        let tip = end;
        let back = end - dir * HEAD_SIZE;
        push_line(verts, tip, back + u * HEAD_SIZE * 0.5, color);
        push_line(verts, tip, back - u * HEAD_SIZE * 0.5, color);
        push_line(verts, tip, back + v * HEAD_SIZE * 0.5, color);
        push_line(verts, tip, back - v * HEAD_SIZE * 0.5, color);
    }
}

fn build_rotate(verts: &mut Vec<LineVertex>, o: Vec3, active: Option<GizmoAxis>) {
    for axis in [GizmoAxis::X, GizmoAxis::Y, GizmoAxis::Z] {
        let color = axis_color(axis, active);
        let (u, v) = ortho_basis(axis.dir());
        let mut prev = o + u * RING_RADIUS;
        for i in 1..=RING_SEGMENTS {
            let theta = (i as f32) * std::f32::consts::TAU / (RING_SEGMENTS as f32);
            let p = o + u * (RING_RADIUS * theta.cos()) + v * (RING_RADIUS * theta.sin());
            push_line(verts, prev, p, color);
            prev = p;
        }
    }
}

fn build_scale(verts: &mut Vec<LineVertex>, o: Vec3, active: Option<GizmoAxis>) {
    for axis in [GizmoAxis::X, GizmoAxis::Y, GizmoAxis::Z] {
        let color = axis_color(axis, active);
        let dir = axis.dir();
        let start = o + dir * AXIS_BASE;
        let end = o + dir * AXIS_LENGTH;
        push_line(verts, start, end, color);
        // Small cube at tip (12 edges).
        let (u, v) = ortho_basis(dir);
        let c = end;
        let p000 = c - u * SCALE_HANDLE - v * SCALE_HANDLE - dir * SCALE_HANDLE;
        let p100 = c + u * SCALE_HANDLE - v * SCALE_HANDLE - dir * SCALE_HANDLE;
        let p110 = c + u * SCALE_HANDLE + v * SCALE_HANDLE - dir * SCALE_HANDLE;
        let p010 = c - u * SCALE_HANDLE + v * SCALE_HANDLE - dir * SCALE_HANDLE;
        let p001 = c - u * SCALE_HANDLE - v * SCALE_HANDLE + dir * SCALE_HANDLE;
        let p101 = c + u * SCALE_HANDLE - v * SCALE_HANDLE + dir * SCALE_HANDLE;
        let p111 = c + u * SCALE_HANDLE + v * SCALE_HANDLE + dir * SCALE_HANDLE;
        let p011 = c - u * SCALE_HANDLE + v * SCALE_HANDLE + dir * SCALE_HANDLE;
        // bottom square
        push_line(verts, p000, p100, color);
        push_line(verts, p100, p110, color);
        push_line(verts, p110, p010, color);
        push_line(verts, p010, p000, color);
        // top square
        push_line(verts, p001, p101, color);
        push_line(verts, p101, p111, color);
        push_line(verts, p111, p011, color);
        push_line(verts, p011, p001, color);
        // vertical edges
        push_line(verts, p000, p001, color);
        push_line(verts, p100, p101, color);
        push_line(verts, p110, p111, color);
        push_line(verts, p010, p011, color);
    }
}

fn ortho_basis(axis: Vec3) -> (Vec3, Vec3) {
    let helper = if axis.y.abs() < 0.9 { Vec3::new(0.0, 1.0, 0.0) } else { Vec3::new(1.0, 0.0, 0.0) };
    let u = axis.cross(helper).normalize();
    let v = axis.cross(u).normalize();
    (u, v)
}
```

```rust
//! Rotate gizmo — ring plane intersection + atan2 angle delta.

use voltex_ecs::Transform;
use voltex_math::{Ray, Vec3};

use super::{DragState, GizmoAxis};

const RING_RADIUS: f32 = 1.2;
const RING_TOLERANCE: f32 = 0.15;

fn ring_plane_normal(axis: GizmoAxis) -> Vec3 { axis.dir() }

fn ortho_basis(axis: Vec3) -> (Vec3, Vec3) {
    let helper = if axis.y.abs() < 0.9 { Vec3::new(0.0, 1.0, 0.0) } else { Vec3::new(1.0, 0.0, 0.0) };
    let u = axis.cross(helper).normalize();
    let v = axis.cross(u).normalize();
    (u, v)
}

fn ray_plane_hit(ray: &Ray, origin: Vec3, normal: Vec3) -> Option<Vec3> {
    let denom = normal.dot(ray.direction);
    if denom.abs() < 1e-6 { return None; }
    let t = (origin - ray.origin).dot(normal) / denom;
    if t < 0.0 { return None; }
    Some(ray.origin + ray.direction * t)
}

fn angle_on_ring(p: Vec3, origin: Vec3, axis: GizmoAxis) -> f32 {
    let (u, v) = ortho_basis(axis.dir());
    let rel = p - origin;
    rel.dot(v).atan2(rel.dot(u))
}

/// Test all three rings; return the axis whose hit lies in the donut
/// (radius ± tolerance), with the smallest ray-t.
pub fn hit_axis(origin: Vec3, ray: &Ray) -> Option<(GizmoAxis, Vec3)> {
    let mut best: Option<(GizmoAxis, f32, Vec3)> = None;
    for axis in [GizmoAxis::X, GizmoAxis::Y, GizmoAxis::Z] {
        let n = ring_plane_normal(axis);
        let hit = match ray_plane_hit(ray, origin, n) { Some(h) => h, None => continue };
        let r = (hit - origin).length();
        if (r - RING_RADIUS).abs() > RING_TOLERANCE { continue; }
        let t = (hit - ray.origin).length();
        best = match best {
            Some((_, bt, _)) if bt <= t => best,
            _ => Some((axis, t, hit)),
        };
    }
    best.map(|(a, _, h)| (a, h))
}

pub fn try_begin(origin: Vec3, ray: &Ray, _cam_pos: Vec3, initial: Transform) -> Option<DragState> {
    let (axis, hit) = hit_axis(origin, ray)?;
    let anchor = angle_on_ring(hit, origin, axis);
    Some(DragState {
        axis,
        initial_transform: initial,
        gizmo_origin: origin,
        plane_normal: ring_plane_normal(axis),
        anchor,
    })
}

pub fn compute_transform(drag: &DragState, ray: &Ray) -> Transform {
    let hit = match ray_plane_hit(ray, drag.gizmo_origin, drag.plane_normal) {
        Some(h) => h,
        None => return drag.initial_transform,
    };
    let cur = angle_on_ring(hit, drag.gizmo_origin, drag.axis);
    let mut delta = cur - drag.anchor;
    if delta > std::f32::consts::PI { delta -= std::f32::consts::TAU; }
    if delta < -std::f32::consts::PI { delta += std::f32::consts::TAU; }
    let mut t = drag.initial_transform;
    let idx = drag.axis.index();
    let mut rot = [t.rotation.x, t.rotation.y, t.rotation.z];
    rot[idx] += delta;
    t.rotation = Vec3::new(rot[0], rot[1], rot[2]);
    t
}
```

```rust
//! Scale gizmo — same hit detection as translate; drag delta becomes a
//! per-axis scale multiplier.

use voltex_ecs::Transform;
use voltex_math::{AABB, Ray, Vec3};

use super::{DragState, GizmoAxis};

const AXIS_LENGTH: f32 = 1.5;
const HANDLE_HALF: f32 = 0.12;
const HANDLE_OFFSET: f32 = AXIS_LENGTH - HANDLE_HALF;
const AXIS_THICKNESS: f32 = 0.08;
const AXIS_BASE_OFFSET: f32 = 0.15;

fn handle_local_aabb(axis: GizmoAxis) -> AABB {
    let dir = axis.dir();
    let center = dir * HANDLE_OFFSET;
    AABB::new(
        center - Vec3::new(HANDLE_HALF, HANDLE_HALF, HANDLE_HALF),
        center + Vec3::new(HANDLE_HALF, HANDLE_HALF, HANDLE_HALF),
    )
}

fn axis_local_aabb(axis: GizmoAxis) -> AABB {
    match axis {
        GizmoAxis::X => AABB::new(
            Vec3::new(AXIS_BASE_OFFSET, -AXIS_THICKNESS, -AXIS_THICKNESS),
            Vec3::new(AXIS_LENGTH, AXIS_THICKNESS, AXIS_THICKNESS),
        ),
        GizmoAxis::Y => AABB::new(
            Vec3::new(-AXIS_THICKNESS, AXIS_BASE_OFFSET, -AXIS_THICKNESS),
            Vec3::new(AXIS_THICKNESS, AXIS_LENGTH, AXIS_THICKNESS),
        ),
        GizmoAxis::Z => AABB::new(
            Vec3::new(-AXIS_THICKNESS, -AXIS_THICKNESS, AXIS_BASE_OFFSET),
            Vec3::new(AXIS_THICKNESS, AXIS_THICKNESS, AXIS_LENGTH),
        ),
    }
}

fn pick_drag_plane_normal(axis: GizmoAxis, gizmo_origin: Vec3, cam_pos: Vec3) -> Vec3 {
    let cam_dir = (cam_pos - gizmo_origin).normalize();
    let (n1, n2) = match axis {
        GizmoAxis::X => (Vec3::new(0.0, 1.0, 0.0), Vec3::new(0.0, 0.0, 1.0)),
        GizmoAxis::Y => (Vec3::new(1.0, 0.0, 0.0), Vec3::new(0.0, 0.0, 1.0)),
        GizmoAxis::Z => (Vec3::new(1.0, 0.0, 0.0), Vec3::new(0.0, 1.0, 0.0)),
    };
    if cam_dir.dot(n1).abs() >= cam_dir.dot(n2).abs() { n1 } else { n2 }
}

fn ray_plane_hit(ray: &Ray, origin: Vec3, normal: Vec3) -> Option<Vec3> {
    let denom = normal.dot(ray.direction);
    if denom.abs() < 1e-6 { return None; }
    let t = (origin - ray.origin).dot(normal) / denom;
    if t < 0.0 { return None; }
    Some(ray.origin + ray.direction * t)
}

pub fn hit_axis(gizmo_origin: Vec3, ray: &Ray) -> Option<GizmoAxis> {
    let mut best: Option<(GizmoAxis, f32)> = None;
    for axis in [GizmoAxis::X, GizmoAxis::Y, GizmoAxis::Z] {
        // Test both the axis bar AND the cube handle at its tip.
        let bar = axis_local_aabb(axis);
        let bar_w = AABB::new(bar.min + gizmo_origin, bar.max + gizmo_origin);
        if let Some(t) = crate::picking::ray_vs_aabb(ray, &bar_w) {
            best = match best {
                Some((_, bt)) if bt <= t => best,
                _ => Some((axis, t)),
            };
        }
        let h = handle_local_aabb(axis);
        let h_w = AABB::new(h.min + gizmo_origin, h.max + gizmo_origin);
        if let Some(t) = crate::picking::ray_vs_aabb(ray, &h_w) {
            best = match best {
                Some((_, bt)) if bt <= t => best,
                _ => Some((axis, t)),
            };
        }
    }
    best.map(|(a, _)| a)
}

pub fn try_begin(origin: Vec3, ray: &Ray, cam_pos: Vec3, initial: Transform) -> Option<DragState> {
    let axis = hit_axis(origin, ray)?;
    let plane_normal = pick_drag_plane_normal(axis, origin, cam_pos);
    let hit = ray_plane_hit(ray, origin, plane_normal)?;
    let anchor = (hit - origin).dot(axis.dir());
    Some(DragState {
        axis,
        initial_transform: initial,
        gizmo_origin: origin,
        plane_normal,
        anchor,
    })
}

pub fn compute_transform(drag: &DragState, ray: &Ray) -> Transform {
    let hit = match ray_plane_hit(ray, drag.gizmo_origin, drag.plane_normal) {
        Some(h) => h,
        None => return drag.initial_transform,
    };
    let current = (hit - drag.gizmo_origin).dot(drag.axis.dir());
    let factor = (current / drag.anchor.max(1e-3)).max(0.01);
    let mut t = drag.initial_transform;
    let idx = drag.axis.index();
    let mut s = [t.scale.x, t.scale.y, t.scale.z];
    s[idx] *= factor;
    t.scale = Vec3::new(s[0], s[1], s[2]);
    t
}
```
